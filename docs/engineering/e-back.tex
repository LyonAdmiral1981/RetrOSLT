\newcommand\gl[2]{\par\noindent\hangindent=1.2em\textbf{#1}\quad #2\par}

\gl{8088}{The Intel processor of the IBM PC and XT: twenty address lines,
1 MB, data eight bits at a time.}
\gl{BPB}{BIOS Parameter Block: the description of a volume in its startup
record.}
\gl{cluster}{The unit in which disk space is given to files.}
\gl{DMA}{Direct memory access: the PC's 8237 controller moves disk data to
and from memory, within one 64 KB page at a time.}
\gl{EXEC}{DOS function 4Bh, which loads and runs a program; in PC DOS 2
part of \file{COMMAND.COM}.}
\gl{FAT}{File allocation table: the map of which clusters belong to which
file; 12-bit entries under DOS 2.}
\gl{FCB}{File control block: DOS 1's way of naming a file to the kernel,
kept in DOS 2; a program's first two are in its PSP.}
\gl{kernel}{\file{RETDOS.COM}: the part of DOS that provides the interrupt
21h functions.}
\gl{kit}{The source Microsoft sent to computer makers, from which each
built its own DOS.}
\gl{listing}{In this book, an excerpt of the source, numbered as in its
file.}
\gl{OEM}{Original equipment manufacturer: in the kit, the maker who adds
the parts for its own computer.}
\gl{patch}{A file of the lines a change adds to and removes from the
source.}
\gl{PSP}{Program Segment Prefix: the 256 bytes DOS puts before every
program.}
\gl{release number}{\code{R} and the date and time of the last change to
RetrOS/LT's own files, such as \release.}
\gl{series}{The list of patches, in order.}
\gl{startup record}{The first sector of a diskette or partition, which
loads \file{RETBIO.COM}; also called the boot sector.}
\gl{SYSINIT}{Microsoft's start-up code, linked into \file{RETBIO.COM},
which reads \file{CONFIG.SYS} and starts \file{COMMAND.COM}.}
