CALCBPB PROC    NEAR
        MOV     [CB_SECS],AX
        MOV     [CB_SPC],1
CB_SPCL:
        MOV     [CB_FAT],1
CB_FATL:
        MOV     AX,[CB_SECS]
        SUB     AX,1+32                 ; boot sector, root directory
        SUB     AX,[CB_FAT]
        SUB     AX,[CB_FAT]
        XOR     DX,DX
        DIV     [CB_SPC]
        MOV     [CB_CLUS],AX
        ADD     AX,2                    ; bytes the FAT needs: (n+2) * 1.5
        MOV     CX,3
        MUL     CX
        SHR     DX,1
        RCR     AX,1
        MOV     CX,[CB_FAT]             ; bytes it has: sectors * 512
        MOV     BX,CX
        MOV     CL,7
        SHR     BX,CL                   ; high word
        MOV     CX,[CB_FAT]
        PUSH    AX
        MOV     AX,CX
        MOV     CL,9
        SHL     AX,CL                   ; low word
        MOV     CX,AX
        POP     AX
        CMP     DX,BX
        JB      CB_FITS
        JA      CB_MORE
        CMP     AX,CX
        JBE     CB_FITS
CB_MORE:
        INC     [CB_FAT]
        JMP     CB_FATL
CB_FITS:
        CMP     [CB_CLUS],4084
        JBE     CB_DONE
        SHL     [CB_SPC],1
        JMP     CB_SPCL
CB_DONE:
        MOV     WORD PTR [CB_BPB],SECSIZE
        MOV     AL,BYTE PTR [CB_SPC]
        MOV     [CB_BPB+2],AL
        MOV     WORD PTR [CB_BPB+3],1
        MOV     [CB_BPB+5],2
        MOV     WORD PTR [CB_BPB+6],512
        MOV     AX,[CB_SECS]
        MOV     WORD PTR [CB_BPB+8],AX
        MOV     [CB_BPB+10],0F8H
        MOV     AX,[CB_FAT]
        MOV     WORD PTR [CB_BPB+11],AX
        RET
CALCBPB ENDP
