        ; a run of clusters one after another on the disk, read at once
        ; (a track at a time) rather than a cluster at a time
        MOV     [LD_RUN],AX
        MOV     [LD_LEN],1
LD_MORE:
        CALL    NEXTCLUS
        JC      LD_ERR
        MOV     AX,[LD_RUN]
        ADD     AX,[LD_LEN]
        CMP     AX,[LD_CLUS]
        JNZ     LD_READ
        INC     [LD_LEN]
        JMP     LD_MORE
LD_READ:
        MOV     AX,[LD_RUN]
        SUB     AX,2
        MOV     CL,[BS_SPC]
        XOR     CH,CH
        MUL     CX
        ADD     AX,[DATALO]
        ADC     DX,0
        PUSH    AX
        PUSH    DX
        MOV     AX,[LD_LEN]
        MUL     CX
        MOV     CX,AX                   ; sectors
        POP     DX
        POP     AX
        CALL    DISKIO
        JC      LD_ERR
        JMP     LD_LP
